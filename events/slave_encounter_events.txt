namespace = slave_encounter
####################################
# 1001 - Garden Encounter
####################################
slave_encounter.1001 = {
	type = character_event
	title = slave_encounter.1001.t
	desc = slave_encounter.1001.desc
	theme = seduction
	left_portrait = {
		character = root
		animation = personality_calm
	}
	right_portrait = {
		trigger = { exists = scope:encounter_slave
		}
		character = scope:encounter_slave
		animation = idle
	}
	trigger = {
		NOT = { has_character_flag = slave_encounter_cooldown
		}
		any_courtier_or_guest = {
			root = { has_relation_slave = prev
			}
			is_alive = yes
		}
	}
	immediate = {
		add_character_flag = {
			flag = slave_encounter_cooldown
			days = 60
		}
		random_courtier_or_guest = {
			limit = {
				root = { has_relation_slave = prev
				}
				is_alive = yes
			}
			weight = {
				base = 10
				modifier = {
					add = 15
					has_character_flag = concubine_slave_flag
				}
				modifier = {
					add = 40
					has_character_flag = secondary_slave_flag
				}
				modifier = {
					add = 90
					has_character_flag = great_slave_flag
				}
			}
			save_scope_as = encounter_slave
		}
	}
	option = {
		name = slave_encounter.1001.a
		add_prestige = 5
		stress_impact = {
			base = -5
		}
	}
	option = {
		name = slave_encounter.1001.b
		scope:encounter_slave = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 10
			}
		}
	}
	option = {
		name = slave_encounter.1001.c
		add_dread = 2
	}
}
