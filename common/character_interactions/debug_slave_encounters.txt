# Debug interactions to force-fire slave encounter events
# Right-click yourself in debug mode to use these
# Fires the full random system (weighted pick + random location)
debug_force_slave_encounter_random = {
	category = debug_main
	common_interaction = yes
	icon = debug_secret
	ignores_pending_interaction_block = yes
	use_diplomatic_range = no
	is_shown = { always = yes
	}
	is_valid = { always = yes
	}
	desc = debug_force_slave_encounter_random_desc
	on_accept = {
		scope:actor = {
			remove_character_flag = slave_encounter_cooldown
			trigger_event = {
				on_action = slave_random_encounter_pulse
			}
		}
	}
	ai_potential = { always = no
	}
}
