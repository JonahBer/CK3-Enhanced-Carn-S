# common/scripted_guis/slave_management_actions.txt
#
# EXACTLY matches debug_interactions tier-set behavior, but on scope:slave:
# - remove the other 3 tier flags
# - add the desired tier flag via add_character_flag = { flag = ... }
#
# Now includes max-count checks per tier. Each promote/demote block verifies
# that the TARGET tier has not reached its cap before allowing the transition.
# Max values are stored as variables on root (the player character):
#   max_decorative_slaves, max_concubine_slaves, max_secondary_slaves, max_great_slaves
# These are initialized in toggle_slave_management_window on window open.
#
# Scopes:
# - ROOT = player (GuiScope.SetRoot(GetPlayer.MakeScope))
# - scope:slave = selected character passed from GUI via AddScope('slave', Character.MakeScope)
slave_mgmt_promote = {
	scope = character
	saved_scopes = { slave
	}
	effect = {
		scope:slave = {
			# ── decorative → concubine ──
			if = {
				limit = {
					has_character_flag = decorative_slave_flag
					root.var:concubine_slaves_count < root.var:max_concubine_slaves
				}
				# EXACTLY debug_set_tier_concubine_slave_flag logic
				remove_character_flag = decorative_slave_flag
				remove_character_flag = secondary_slave_flag
				remove_character_flag = great_slave_flag
				add_character_flag = { flag = concubine_slave_flag
				}
				set_variable = {
					name = status_tier
					value = 2000
				}
				root = {
					remove_list_variable = {
						name = decorative_slaves
						target = prev
					}
					add_to_variable_list = {
						name = concubine_slaves
						target = prev
					}
					change_variable = {
						name = decorative_slaves_count
						add = -1
					}
					change_variable = {
						name = concubine_slaves_count
						add = 1
					}
				}
			}
			# ── concubine → secondary ──
			else_if = {
				limit = {
					has_character_flag = concubine_slave_flag
					root.var:secondary_slaves_count < root.var:max_secondary_slaves
				}
				# EXACTLY debug_set_tier_secondary_slave_flag logic
				remove_character_flag = decorative_slave_flag
				remove_character_flag = concubine_slave_flag
				remove_character_flag = great_slave_flag
				add_character_flag = { flag = secondary_slave_flag
				}
				set_variable = {
					name = status_tier
					value = 3000
				}
				root = {
					remove_list_variable = {
						name = concubine_slaves
						target = prev
					}
					add_to_variable_list = {
						name = secondary_slaves
						target = prev
					}
					change_variable = {
						name = concubine_slaves_count
						add = -1
					}
					change_variable = {
						name = secondary_slaves_count
						add = 1
					}
				}
			}
			# ── secondary → great ──
			else_if = {
				limit = {
					has_character_flag = secondary_slave_flag
					root.var:great_slaves_count < root.var:max_great_slaves
				}
				# EXACTLY debug_set_tier_great_slave_flag logic
				remove_character_flag = decorative_slave_flag
				remove_character_flag = concubine_slave_flag
				remove_character_flag = secondary_slave_flag
				add_character_flag = { flag = great_slave_flag
				}
				set_variable = {
					name = status_tier
					value = 4000
				}
				root = {
					remove_list_variable = {
						name = secondary_slaves
						target = prev
					}
					add_to_variable_list = {
						name = great_slaves
						target = prev
					}
					change_variable = {
						name = secondary_slaves_count
						add = -1
					}
					change_variable = {
						name = great_slaves_count
						add = 1
					}
				}
			}
		}
	}
}
